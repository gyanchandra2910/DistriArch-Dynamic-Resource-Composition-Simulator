cmake_minimum_required(VERSION 3.20)
project(DistriArch VERSION 3.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
add_library(distriarch_core src/model.cpp src/engine.cpp src/metrics.cpp src/experiments.cpp)
target_include_directories(distriarch_core PUBLIC include third_party)
target_link_libraries(distriarch_core PUBLIC ZLIB::ZLIB Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(distriarch_core PRIVATE -Wall -Wextra -Wpedantic -Wno-misleading-indentation -ffp-contract=off)
endif()
add_executable(distriarch src/main.cpp src/server.cpp)
target_link_libraries(distriarch PRIVATE distriarch_core)
if(WIN32)
  target_compile_definitions(distriarch PRIVATE _WIN32_WINNT=0x0A00)
  target_link_libraries(distriarch PRIVATE ws2_32 crypt32)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(distriarch_tests tests/native_tests.cpp)
  target_link_libraries(distriarch_tests PRIVATE distriarch_core)
  add_test(NAME model COMMAND distriarch_tests --model)
  add_test(NAME integrity COMMAND distriarch_tests --integrity)
  set_tests_properties(model integrity PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
